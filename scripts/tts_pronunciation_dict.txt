# Единый словарь произношения для TTS (Yandex SpeechKit / Silero).
# Формат: слово = замена  (регистр ключа не важен)
# Применяется ПОСЛЕ авто-ударений; ключ матчится и к формам с + / ́ .
#
# Типы замен:
#   1) alias — обычный текст / расшифровка
#   2) stress — форма с + перед ударной гласной (Yandex/Silero)
#   3) phoneme — блок [[...]] (только Yandex TTS markup)
#
# Источник ошибок: ручной QA озвучки → одна строка сюда.

# --- г→в баг SpeechKit: «много» → «мново» ---
# Ударная «о» = ɔ, безударная «а» = ʌ; согласная жёстко g (не v).
много = [[m n ɔ g ʌ]]
немного = [[nʲ ɪ m n ɔ g ʌ]]
многое = [[m n ɔ g ʌ j ɪ]]
многого = [[m n ɔ g ʌ v ʌ]]
многому = [[m n ɔ g ʌ m ʊ]]
многом = [[m n ɔ g ʌ m]]
многим = [[m n ɔ gʲ ɪ m]]
многих = [[m n ɔ gʲ ɪ x]]
многими = [[m n ɔ gʲ ɪ mʲ ɪ]]

# --- домен покер (бывший tts_custom_dict_poker.txt) ---
покерное = п+окерное
мышление = мышл+ение
покерист = покер+ист
покериста = покер+иста
покеристу = покер+исту
покеристом = покер+истом
покеристы = покер+исты

# --- омонимы замок/мука/стоит — в tts_disambig.py (контекст), не плоским k=v ---

# --- от чег+о (не «О́тчего» / не отче́го) ---
от чего = от чег+о
отчего = от чег+о
для чего = для чег+о
про чего = про чег+о

самоодобрению = самоодобр+ению
самоодобрением = самоодобр+ением
самоодобрении = самоодобр+ении

дорого = д+орого

дорогого = д+орогого

# --- сленг / жанры ---
нищепанк = нищеп+анк
нищепанка = нищеп+анка
нищепанке = нищеп+анке
нищепанком = нищеп+анком
нищепанку = нищеп+анку


# --- о́рган (муз.) vs орга́н (тело/власть) ---
# Контекст [after:…] — слово СЛЕВА. Голое «орган» не трогаем.
орган [after:звук|музыкальный|трубный|церковный|электронный] = +орган
органа [after:звук|музыка|игра|тембр] = +органа
орган [after:половой|внутренний|государственный|исполнительный] = орг+ан
органа [after:полового|внутреннего|государственного] = орг+ана
орган [after:власти|чувств] = орг+ан
# фразы (на случай «орган власти» без падежной формы в after)
орган власти = орг+ан власти
орган чувств = орг+ан чувств


# --- стороны́ (род.п.) vs сто́роны (им.п. мн.) ---
